module Stage12.MultipleViews

bracedMulti x y with (view1 x) | (view2 y) { bracedMulti x y | True | False = 1; bracedMulti x y | _ | Just z = z; }

layoutMulti x y with (view1 x) | (view2 y)
  layoutMulti x y | True | False = 1
  layoutMulti x y | False | True = 2
  layoutMulti x y | _ | _ = 0

allLengths xs ys with (isNil xs) | (isNil ys)
  allLengths xs ys | True | True = 0
  allLengths xs ys | _ | _ = 1

nextDecl : Nat
nextDecl = 42
