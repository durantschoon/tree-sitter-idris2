module Stage11.LayoutWith

check n with (isZero n)
  check n | True = 1

parity n with (isEven n)
  parity n | True = 0
  parity n | False = 1

mixed value with (view value)
  mixed value | Just x = x
  mixed _ | Nothing = 0
  mixed value | False = case value of { Nothing => 0; Just y => y }

nested value with (case value of { Nothing => 0; Just x => x })
  _ | 0 = case value of { Nothing => 0; Just x => x }
  nested value | _ = value + 1

followWith x with (isZero x)
  followWith x | True = 1
  followWith x | False = 0

nextDecl : Nat
nextDecl = 42
